<script>
    // Cole aqui o link da sua implantação do Google Apps Script
    const SCRIPT_URL = "SUA_URL_AQUI"; 
    
    let usuarioLogado = "";
    let cargoLogado = "player"; // player ou admin
    let cacheDados = {};

    function logar() {
        const u = document.getElementById('username').value.trim();
        const p = document.getElementById('password').value;
        const l = document.getElementById('team-link').value.trim();

        if(!u || !p) return alert("Preencha os campos obrigatórios.");

        fetch(SCRIPT_URL, {
            method: "POST",
            body: JSON.stringify({ action: "registrarOuLogar", username: u, password: p, teamLink: l })
        })
        .then(res => res.json())
        .then(res => {
            if(res.status === "success") {
                usuarioLogado = u;
                cargoLogado = res.role;
                document.getElementById('user-display').innerText = u;
                document.getElementById('auth-screen').classList.add('hidden');
                document.getElementById('main-screen').classList.remove('hidden');
                
                if(cargoLogado === "admin") {
                    document.getElementById('admin-panel').classList.remove('hidden');
                }
                carregarDadosDoBanco();
            } else {
                alert(res.msg);
            }
        });
    }

    function carregarDadosDoBanco() {
        fetch(SCRIPT_URL).then(res => res.json()).then(data => {
            cacheDados = data;
            renderizarTabelas();
        });
    }

    function renderizarTabelas() {
        // Renderiza Classificação
        const sBody = document.querySelector('#standings-table tbody');
        sBody.innerHTML = "";
        let sorted = cacheDados.jogadores.sort((a,b) => b.pontos - a.pontos);
        sorted.forEach((j, i) => {
            sBody.innerHTML += `<tr><td>${i+1}º</td><td><b>${j.nome}</b></td><td>${j.pontos}</td><td>${j.status}</td></tr>`;
        });

        // Renderiza Times baseado nas regras de sigilo solicitadas
        const tBody = document.querySelector('#teams-table tbody');
        const msgBlock = document.getElementById('block-message');
        tBody.innerHTML = "";
        
        if(cacheDados.config.listaAberta === "fechada" && cargoLogado !== "admin") {
            msgBlock.classList.remove('hidden');
        } else {
            msgBlock.classList.add('hidden');
            cacheDados.jogadores.forEach(j => {
                let linkHtml = j.time.startsWith('http') ? `<a href="${j.time}" target="_blank" class="link-btn">Ver Time 🔗</a>` : 'Não enviado';
                tBody.innerHTML += `<tr><td>${j.nome}</td><td>${linkHtml}</td></tr>`;
            });
        }
    }

    function definirLista(estado) {
        fetch(SCRIPT_URL, { method: "POST", body: JSON.stringify({ action: "toggleLista", estado: estado }) })
        .then(() => { alert("Estado da lista modificado!"); carregarDadosDoBanco(); });
    }

    function resetarTudo() {
        if(confirm("Tem certeza que deseja apagar TODOS OS DADOS do torneio?")) {
            fetch(SCRIPT_URL, { method: "POST", body: JSON.stringify({ action: "resetTorneio" }) })
            .then(() => { alert("Torneio Resetado!"); window.location.reload(); });
        }
    }

    function mudarAba(aba) {
        document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
        document.getElementById('content-home').classList.add('hidden');
        document.getElementById('content-teams').classList.add('hidden');
        document.getElementById('content-matches').classList.add('hidden');
        document.getElementById('tab-'+aba).classList.add('active');
        document.getElementById('content-'+aba).classList.remove('hidden');
    }
</script>
